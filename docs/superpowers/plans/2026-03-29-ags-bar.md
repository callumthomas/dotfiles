# AGS v2 Bar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Waybar with an AGS v2 (Astal) bar featuring interactive dropdown panels for all modules.

**Architecture:** Widget-per-module with a shared popup manager. Each module exports a bar button and its dropdown panel as TSX components. A singleton PopupManager coordinates which popup is open. The bar is a layer-shell window on each monitor; popups are separate layer-shell windows positioned below their trigger.

**Tech Stack:** AGS v2, Astal libraries, TypeScript + JSX (GTK4), GTK4 CSS

**Spec:** `docs/superpowers/specs/2026-03-29-ags-bar-design.md`

### Task 4: Popup Manager & PopupWindow Component

**Files:**
- Create: `ags/bar/PopupManager.ts`
- Create: `ags/components/PopupWindow.tsx`

**Docs to reference:**
- Astal4 Window API: https://aylur.github.io/libastal/astal4/class.Window.html
- Popup pattern from AGS examples

The popup system uses one layer-shell window per popup. PopupManager tracks which is open and handles closing. PopupWindow is a reusable wrapper that creates a layer-shell window with a click-outside backdrop.

- [ ] **Step 1: Write PopupManager.ts**

```typescript
// Singleton — tracks currently open popup name, ensures only one is open at a time

import { App } from "ags/gtk4";

let currentPopup: string | null = null;

export function openPopup(name: string) {
  if (currentPopup && currentPopup !== name) {
    closePopup();
  }
  const win = App.get_window(name);
  if (win) {
    win.visible = true;
    currentPopup = name;
  }
}

export function closePopup() {
  if (currentPopup) {
    const win = App.get_window(currentPopup);
    if (win) {
      win.visible = false;
    }
    currentPopup = null;
  }
}

export function togglePopup(name: string) {
  if (currentPopup === name) {
    closePopup();
  } else {
    openPopup(name);
  }
}
```

- [ ] **Step 2: Write PopupWindow.tsx**

```tsx
import Astal from "gi://Astal?version=4.0";
import { closePopup } from "../bar/PopupManager";

interface PopupWindowProps {
  name: string;
  gdkmonitor: Gdk.Monitor;
  children?: JSX.Element;
}

export default function PopupWindow({ name, gdkmonitor, children }: PopupWindowProps) {
  return (
    <window
      name={name}
      gdkmonitor={gdkmonitor}
      anchor={Astal.WindowAnchor.TOP | Astal.WindowAnchor.RIGHT}
      exclusivity={Astal.Exclusivity.IGNORE}
      layer={Astal.Layer.TOP}
      keymode={Astal.Keymode.ON_DEMAND}
      visible={false}
      cssClasses={["popup-window"]}
      onKeyPressed={(self, keyval) => {
        if (keyval === Gdk.KEY_Escape) {
          closePopup();
        }
      }}
    >
      <eventbox onClick={() => closePopup()}>
        <box vertical>
          <box cssClasses={["popup-panel"]}>
            {children}
          </box>
        </box>
      </eventbox>
    </window>
  );
}
```

Note: The exact popup positioning (below the trigger button) may need adjustment based on AGS v2's layout behavior. The popup anchors to TOP | RIGHT which places it near the right side of the bar. Per-module horizontal offset can be handled with margins in CSS or by adjusting the anchor per popup.

### Task 6: Window Title Widget

- [ ] **Step 1: Write window-title.tsx**

```tsx
import Hyprland from "gi://AstalHyprland";
import { bind } from "ags/gtk4";

const hyprland = Hyprland.get_default();

const rewriteRules: [RegExp, string][] = [
  [/^(.*) - Mozilla Firefox$/, "🌎 $1"],
  [/^(.*) — Mozilla Firefox$/, "🌎 $1"],
  [/^(.*) - Delio - Slack$/, "💬 $1"],
  [/^\s*$/, "  ...  "],
];

function rewriteTitle(title: string): string {
  for (const [pattern, replacement] of rewriteRules) {
    if (pattern.test(title)) {
      return title.replace(pattern, replacement);
    }
  }
  return title;
}

function truncate(str: string, max: number): string {
  return str.length > max ? str.slice(0, max - 1) + "…" : str;
}

export default function WindowTitle() {
  return (
    <box cssClasses={["window-title"]}>
      <label
        label={bind(hyprland, "focusedClient").as((client) => {
          const title = client?.title ?? "";
          return truncate(rewriteTitle(title), 50);
        })}
      />
    </box>
  );
}
```

```typescript
interface TlpStatus {
  mode: "AC" | "battery";
  cpuGovernor: string;
  cpuBoost: boolean;
  platformProfile: string;
  startThreshold: number;
  stopThreshold: number;
}

// Poll TLP status every 10 seconds
export const tlpStatus = Variable<TlpStatus>({
  mode: "AC",
  cpuGovernor: "unknown",
  cpuBoost: false,
  platformProfile: "unknown",
  startThreshold: 75,
  stopThreshold: 80,
}).poll(10000, ["bash", "-c", `
  echo "MODE=$(tlp-stat -s 2>/dev/null | grep 'Mode' | awk '{print $NF}')"
  echo "GOV=$(cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_governor 2>/dev/null)"
  echo "BOOST=$(cat /sys/devices/system/cpu/cpufreq/boost 2>/dev/null)"
  echo "PLATFORM=$(cat /sys/firmware/acpi/platform_profile 2>/dev/null)"
  echo "START=$(cat /sys/class/power_supply/BAT0/charge_control_start_threshold 2>/dev/null || echo 75)"
  echo "STOP=$(cat /sys/class/power_supply/BAT0/charge_control_end_threshold 2>/dev/null || echo 80)"
`], (out) => {
  const vars: Record<string, string> = {};
  for (const line of out.split("\n")) {
    const [key, val] = line.split("=");
    if (key && val) vars[key] = val;
  }
  return {
    mode: (vars.MODE?.toLowerCase().includes("bat") ? "battery" : "AC") as "AC" | "battery",
    cpuGovernor: vars.GOV ?? "unknown",
    cpuBoost: vars.BOOST === "1",
    platformProfile: vars.PLATFORM ?? "unknown",
    startThreshold: parseInt(vars.START ?? "75"),
    stopThreshold: parseInt(vars.STOP ?? "80"),
  };
});

// Power draw from BAT0
export const powerDraw = Variable("--").poll(5000, ["bash", "-c",
  "cat /sys/class/power_supply/BAT0/power_now 2>/dev/null || echo 0"
], (out) => {
  const microwatts = parseInt(out.trim());
  const watts = microwatts / 1000000;
  return watts > 0 ? `${watts.toFixed(1)}W` : "--";
});

// Toggle charge thresholds to full (100/100) or back to normal (75/80)
export async function toggleFullCharge() {
  const status = tlpStatus.get();
  if (status.stopThreshold >= 100) {
    // Restore normal thresholds
    await execAsync(["bash", "-c",
      "echo 75 | pkexec tee /sys/class/power_supply/BAT0/charge_control_start_threshold && " +
      "echo 80 | pkexec tee /sys/class/power_supply/BAT0/charge_control_end_threshold"
    ]);
  } else {
    // Set full charge
    await execAsync(["bash", "-c",
      "echo 0 | pkexec tee /sys/class/power_supply/BAT0/charge_control_start_threshold && " +
      "echo 100 | pkexec tee /sys/class/power_supply/BAT0/charge_control_end_threshold"
    ]);
  }
}
```

```tsx
import { Variable, bind, execAsync } from "ags/gtk4";
import { togglePopup } from "../bar/PopupManager";
import PopupWindow from "../components/PopupWindow";

// CPU usage polling
const cpuUsage = Variable(0).poll(2000, ["bash", "-c",
  "head -1 /proc/stat | awk '{idle=$5; total=0; for(i=2;i<=NF;i++) total+=$i; print int(100*(total-idle)/total)}'"
], (out) => parseInt(out.trim()) || 0);

// Memory usage polling
const memInfo = Variable({ used: "0G", total: "0G", percent: 0 }).poll(2000, ["bash", "-c",
  "free -b | awk '/Mem:/ {printf \"%d %d\", $3, $2}'"
], (out) => {
  const [used, total] = out.trim().split(" ").map(Number);
  return {
    used: `${(used / 1073741824).toFixed(1)}G`,
    total: `${(total / 1073741824).toFixed(1)}G`,
    percent: Math.round((used / total) * 100),
  };
});

// Temperature polling
const cpuTemp = Variable(0).poll(5000, ["bash", "-c",
  "sensors | awk '/Tctl:/ {gsub(/[+°C]/, \"\", $2); print int($2)}'"
], (out) => parseInt(out.trim()) || 0);

const temps = Variable<{ name: string; value: number }[]>([]).poll(5000, ["bash", "-c",
  "sensors -j 2>/dev/null"
], (out) => {
  try {
    const data = JSON.parse(out);
    const results: { name: string; value: number }[] = [];
    for (const [chip, sensors] of Object.entries(data as Record<string, any>)) {
      let chipName = chip;
      if (chip.includes("k10temp")) chipName = "CPU";
      else if (chip.includes("amdgpu")) chipName = "GPU";
      else if (chip.includes("nvme")) chipName = "NVMe";
      else continue;

      for (const [sensor, values] of Object.entries(sensors as Record<string, any>)) {
        if (typeof values !== "object" || !values) continue;
        for (const [key, val] of Object.entries(values as Record<string, any>)) {
          if (key.startsWith("temp") && key.endsWith("_input") && typeof val === "number") {
            const label = sensor.replace(":", "");
            results.push({ name: `${chipName} ${label}`, value: Math.round(val) });
          }
        }
      }
    }
    return results;
  } catch {
    return [];
  }
});

// Top processes (polled less frequently)
const topCpu = Variable<{ name: string; usage: number }[]>([]).poll(3000, ["bash", "-c",
  "ps aux --sort=-%cpu | head -4 | tail -3 | awk '{cmd=$11; gsub(/.*\\//, \"\", cmd); printf \"%s %d\\n\", cmd, $3}'"
], (out) =>
  out.trim().split("\n").filter(Boolean).map((line) => {
    const parts = line.split(" ");
    return { name: parts[0], usage: parseInt(parts[1]) || 0 };
  })
);

const topMem = Variable<{ name: string; mem: string }[]>([]).poll(3000, ["bash", "-c",
  "ps aux --sort=-rss | head -4 | tail -3 | awk '{cmd=$11; gsub(/.*\\//, \"\", cmd); rss=$6; if(rss>1048576) printf \"%s %.1fG\\n\", cmd, rss/1048576; else printf \"%s %.0fM\\n\", cmd, rss/1024}'"
], (out) =>
  out.trim().split("\n").filter(Boolean).map((line) => {
    const parts = line.split(" ");
    return { name: parts[0], mem: parts[1] };
  })
);

function statColorClass(value: number, thresholds = [45, 60, 75, 90]): string[] {
  if (value >= thresholds[3]) return ["color-critical"];
  if (value >= thresholds[2]) return ["color-warning"];
  if (value >= thresholds[1]) return ["color-high"];
  if (value >= thresholds[0]) return ["color-moderate"];
  return ["color-normal"];
}

function tempColorClass(value: number): string[] {
  return statColorClass(value, [50, 60, 70, 85]);
}

export function CpuButton() {
  return (
    <button
      cssClasses={["module-button"]}
      onClicked={() => togglePopup("stats-popup")}
    >
      <label label="" cssClasses={bind(cpuUsage).as((v) => statColorClass(v))} />
    </button>
  );
}

export function MemButton() {
  return (
    <button
      cssClasses={["module-button"]}
      onClicked={() => togglePopup("stats-popup")}
    >
      <label label="" cssClasses={bind(memInfo).as((m) => statColorClass(m.percent))} />
    </button>
  );
}

export function TempButton() {
  return (
    <button
      cssClasses={["module-button"]}
      onClicked={() => togglePopup("stats-popup")}
    >
      <label label="" cssClasses={bind(cpuTemp).as((t) => tempColorClass(t))} />
    </button>
  );
}

export function StatsPopup(gdkmonitor: Gdk.Monitor) {
  return (
    <PopupWindow name="stats-popup" gdkmonitor={gdkmonitor}>
      <box vertical>
        {/* CPU */}
        <box>
          <label label="CPU" cssClasses={["popup-section-label"]} hexpand xalign={0} />
          <label
            label={bind(cpuUsage).as((v) => `${v}%`)}
            cssClasses={bind(cpuUsage).as((v) => statColorClass(v))}
          />
        </box>
        <scale value={bind(cpuUsage).as((v) => v / 100)} min={0} max={1} sensitive={false} />
        {bind(topCpu).as((procs) =>
          procs.map((p) => (
            <box>
              <label label={p.name} cssClasses={["text-dim"]} hexpand xalign={0} />
              <label label={`${p.usage}%`} cssClasses={statColorClass(p.usage)} />
            </box>
          ))
        )}

        <box cssClasses={["popup-divider"]} />

        {/* Memory */}
        <box>
          <label label="MEMORY" cssClasses={["popup-section-label"]} hexpand xalign={0} />
          <label
            label={bind(memInfo).as((m) => `${m.used} / ${m.total}`)}
            cssClasses={bind(memInfo).as((m) => statColorClass(m.percent))}
          />
        </box>
        <scale value={bind(memInfo).as((m) => m.percent / 100)} min={0} max={1} sensitive={false} />
        {bind(topMem).as((procs) =>
          procs.map((p) => (
            <box>
              <label label={p.name} cssClasses={["text-dim"]} hexpand xalign={0} />
              <label label={p.mem} />
            </box>
          ))
        )}

        <box cssClasses={["popup-divider"]} />

        {/* Temperatures */}
        <label label="TEMPERATURES" cssClasses={["popup-section-label"]} xalign={0} />
        {bind(temps).as((t) =>
          t.map((sensor) => (
            <box>
              <label label={sensor.name} cssClasses={["text-secondary"]} hexpand xalign={0} />
              <label label={`${sensor.value}°C`} cssClasses={tempColorClass(sensor.value)} />
            </box>
          ))
        )}
      </box>
    </PopupWindow>
  );
}
```

```tsx
const mpris = Mpris.get_default();

export default function MediaPlayer() {
  return (
    <box
      visible={bind(mpris, "players").as((players) => players.length > 0)}
      cssClasses={["media-widget"]}
    >
      {bind(mpris, "players").as((players) => {
        const player = players[0];
        if (!player) return <box />;

        return (
          <box>
            <button cssClasses={["media-button"]} onClicked={() => player.previous()}>
              <label label="⏮" />
            </button>
            <button
              cssClasses={bind(player, "playbackStatus").as((status) =>
                status === Mpris.PlaybackStatus.PLAYING
                  ? ["media-button", "playing"]
                  : ["media-button"]
              )}
              onClicked={() => player.play_pause()}
            >
              <label
                label={bind(player, "playbackStatus").as((status) =>
                  status === Mpris.PlaybackStatus.PLAYING ? "⏸" : "⏵"
                )}
              />
            </button>
            <button cssClasses={["media-button"]} onClicked={() => player.next()}>
              <label label="⏭" />
            </button>
            <label
              label={bind(player, "title").as((title) => {
                const artist = player.artist;
                const display = artist ? `${artist} — ${title}` : title;
                return display.length > 40 ? display.slice(0, 39) + "…" : display;
              })}
              cssClasses={["media-title"]}
            />
          </box>
        );
      })}
    </box>
  );
}
```

```tsx
import Notifd from "gi://AstalNotifd";
import { bind, Variable } from "ags/gtk4";
import { togglePopup, closePopup } from "../bar/PopupManager";
import PopupWindow from "../components/PopupWindow";
import ToggleRow from "../components/ToggleRow";

const notifd = Notifd.get_default();
const dnd = Variable(false);

function timeAgo(timestamp: number): string {
  const seconds = Math.floor(Date.now() / 1000 - timestamp);
  if (seconds < 60) return "now";
  if (seconds < 3600) return `${Math.floor(seconds / 60)}m`;
  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h`;
  return `${Math.floor(seconds / 86400)}d`;
}

export function NotificationButton() {
  return (
    <button
      cssClasses={["module-button"]}
      onClicked={() => togglePopup("notification-popup")}
    >
      <box>
        <label label="󰂚" />
        <label
          label={bind(notifd, "notifications").as((n) =>
            n.length > 0 ? ` ${n.length}` : ""
          )}
          visible={bind(notifd, "notifications").as((n) => n.length > 0)}
        />
      </box>
    </button>
  );
}

export function NotificationPopup(gdkmonitor: Gdk.Monitor) {
  return (
    <PopupWindow name="notification-popup" gdkmonitor={gdkmonitor}>
      <box vertical>
        <box>
          <label label="NOTIFICATIONS" cssClasses={["popup-section-label"]} hexpand xalign={0} />
          <button
            cssClasses={["text-dim"]}
            onClicked={() => {
              for (const n of notifd.get_notifications()) {
                n.dismiss();
              }
            }}
          >
            <label label="Clear all" />
          </button>
          <ToggleRow
            label="🔕 DND"
            active={bind(dnd)}
            onToggled={(v) => dnd.set(v)}
          />
        </box>

        {bind(notifd, "notifications").as((notifications) => {
          if (notifications.length === 0) {
            return (
              <box cssClasses={["popup-item"]}>
                <label label="No notifications" cssClasses={["text-dim"]} />
              </box>
            );
          }

          return (
            <box vertical>
              {notifications
                .sort((a, b) => b.time - a.time)
                .map((notif) => (
                  <box cssClasses={["notification-item"]}>
                    <box vertical hexpand>
                      <label label={notif.appName || "Unknown"} xalign={0} />
                      <label
                        label={notif.summary}
                        cssClasses={["text-secondary"]}
                        xalign={0}
                        wrap
                      />
                    </box>
                    <label label={timeAgo(notif.time)} cssClasses={["text-dim"]} />
                  </box>
                ))}
            </box>
          );
        })}
      </box>
    </PopupWindow>
  );
}

// Toast popup for incoming notifications (shown briefly on screen)
export function NotificationToast(gdkmonitor: Gdk.Monitor) {
  // Connect to the "notified" signal to show a brief popup
  // This is a separate layer-shell window that appears on new notifications
  // and auto-hides after ~5 seconds. Respects DND state.
  //
  // Implementation: listen to notifd.connect("notified", (_, id) => { ... })
  // Show a small popup window, auto-hide with GLib.timeout_add.
  // Skip if dnd.get() is true.
  //
  // The exact implementation depends on AGS v2's signal connection pattern.
  // The structure is:
  // 1. Create a hidden layer-shell window at TOP | RIGHT
  // 2. On "notified" signal, populate with notification content, show window
  // 3. Set timeout to hide after 5 seconds
  // 4. Clicking the toast dismisses it
  //
  // This can be refined during implementation.
  return <box />;
}
```
